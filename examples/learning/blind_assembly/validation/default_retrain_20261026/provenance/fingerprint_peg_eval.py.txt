"""Read-only reset and first-control fingerprints around the native evaluator."""
from __future__ import annotations
import argparse
import hashlib
import json
from pathlib import Path
import runpy
import sys
from typing import Any
import numpy as np
import torch
from embodichain.lab.gym.envs.assembly_env import BlindAssemblyEnv

__all__: list[str] = []

parser=argparse.ArgumentParser(add_help=False)
parser.add_argument('--fingerprint',type=Path,required=True)
opts,remaining=parser.parse_known_args()
output=opts.fingerprint.resolve();output.mkdir(parents=True,exist_ok=True)
assert not (output/'states.npz').exists()
arrays:dict[str,np.ndarray]={};reset_count=0;control_count=0;active=False
original_reset=BlindAssemblyEnv.reset
original_step=BlindAssemblyEnv.step

def _put(key:str,value:Any)->None:
    if isinstance(value,torch.Tensor):
        arrays[key]=value.detach().cpu().contiguous().numpy().copy()
    elif isinstance(value,dict):
        for name,item in sorted(value.items()):_put(key+'/'+str(name),item)

def _state(env:Any,prefix:str)->None:
    _put(prefix+'/qpos',env.robot.get_qpos())
    _put(prefix+'/qvel',env.robot.get_qvel())
    _put(prefix+'/part',env.sim.get_rigid_object(env._moving_uid).get_local_pose(to_matrix=True))
    _put(prefix+'/fixture',env.sim.get_rigid_object(env._fixture_uid).get_local_pose(to_matrix=True))
    _put(prefix+'/elapsed',env._elapsed_steps)
    cached=getattr(env,'_assembly_frame_cache',None)
    if cached is not None:_put(prefix+'/cached_frame',cached[1])

def _reset(env:Any,*args:Any,**kwargs:Any)->Any:
    global reset_count,control_count,active
    result=original_reset(env,*args,**kwargs)
    if kwargs.get('seed')==10000 and env.num_envs==64:
        prefix=f'reset_{reset_count:02d}'
        _state(env,prefix)
        _put(prefix+'/observation',result[0])
        reset_count+=1;control_count=0;active=True
    return result

def _step(env:Any,action:Any)->Any:
    global control_count
    if active and control_count<20:
        prefix=f'reset_{reset_count-1:02d}/before_control_{control_count+1:02d}'
        _state(env,prefix);_put(prefix+'/action',action);control_count+=1
    return original_step(env,action)

BlindAssemblyEnv.reset=_reset
BlindAssemblyEnv.step=_step
sys.argv=[sys.argv[0],*remaining]
status='failed'
try:
    runpy.run_module('scripts.benchmark.assembly.eval_peg',run_name='__main__')
    assert reset_count>=1 and control_count==20, (reset_count,control_count)
    status='completed'
finally:
    BlindAssemblyEnv.reset=original_reset;BlindAssemblyEnv.step=original_step
    np.savez_compressed(output/'states.npz',**arrays)
    metadata={'status':status,'reset_calls_captured':reset_count,'first_controls_captured_after_last_reset':control_count,'scope':'Managed-state CPU copies after explicitly seeded resets and before first20controls. No additional physics or observation-manager calls. Diagnostics not used for model selection or independent-test scoring.','arrays':{k:{'shape':list(a.shape),'dtype':str(a.dtype),'sha256':hashlib.sha256(a.tobytes()).hexdigest()} for k,a in arrays.items()}}
    (output/'fingerprints.json').write_text(json.dumps(metadata,indent=2)+'\n')
